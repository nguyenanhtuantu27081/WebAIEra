(function () {
  var root = document.currentScript.previousElementSibling;
  var cv = root.querySelector('.u226-cv'), cx = cv.getContext('2d');
  var rm = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var W = 340, H = 340, C = 34, dpr = Math.min(2, window.devicePixelRatio || 1), B = [], acc = 0;
  cv.width = W * dpr; cv.height = H * dpr; cx.scale(dpr, dpr);
  var COL = ['129,140,248', '45,212,191', '244,114,182'];
  function spawn() {
    var v = Math.random() < 0.5, line = (1 + Math.floor(Math.random() * ((v ? W : H) / C - 1))) * C - 1, dir = Math.random() < 0.5 ? 1 : -1;
    B.push({ v: v, l: line, p: dir > 0 ? -60 : (v ? H : W) + 60, d: dir, sp: 110 + Math.random() * 90, len: 50 + Math.random() * 40, c: COL[Math.floor(Math.random() * 3)] });
  }
  function draw() {
    cx.clearRect(0, 0, W, H);
    B.forEach(function (b) {
      var x0 = b.v ? b.l : b.p - b.d * b.len, y0 = b.v ? b.p - b.d * b.len : b.l, x1 = b.v ? b.l : b.p, y1 = b.v ? b.p : b.l;
      var g = cx.createLinearGradient(x0, y0, x1, y1); g.addColorStop(0, 'rgba(' + b.c + ',0)'); g.addColorStop(1, 'rgba(' + b.c + ',.95)');
      cx.strokeStyle = g; cx.lineWidth = 1.6; cx.beginPath(); cx.moveTo(x0 + 0.5, y0 + 0.5); cx.lineTo(x1 + 0.5, y1 + 0.5); cx.stroke();
      cx.fillStyle = 'rgba(' + b.c + ',1)'; cx.shadowColor = 'rgba(' + b.c + ',1)'; cx.shadowBlur = 8; cx.beginPath(); cx.arc(x1 + 0.5, y1 + 0.5, 1.8, 0, 6.2832); cx.fill(); cx.shadowBlur = 0;
    });
  }
  var raf = 0, last = 0, vis = true;
  function frame(now) {
    raf = 0;
    if (!root.isConnected || !vis) return;
    var dt = Math.min(0.05, Math.max(0, (now - last) / 1000)); last = now;
    acc += dt; if (acc > 0.42 && B.length < 9) { acc = 0; spawn(); }
    B.forEach(function (b) { b.p += b.d * b.sp * dt; });
    B = B.filter(function (b) { var lim = (b.v ? H : W) + 70; return b.d > 0 ? b.p < lim : b.p > -70; });
    draw();
    raf = requestAnimationFrame(frame);
  }
  function run() { if (!raf && vis && !rm) { last = performance.now(); raf = requestAnimationFrame(frame); } }
  if ('IntersectionObserver' in window) new IntersectionObserver(function (es) { vis = es[0].isIntersecting; run(); }).observe(root);
  root.addEventListener('pointermove', function (e) { var r = root.getBoundingClientRect(), k = W / r.width; root.style.setProperty('--mx', ((e.clientX - r.left) * k).toFixed(0) + 'px'); root.style.setProperty('--my', ((e.clientY - r.top) * k).toFixed(0) + 'px'); });
  root.addEventListener('pointerleave', function () { root.style.setProperty('--mx', '-200px'); root.style.setProperty('--my', '-200px'); });
  for (var i = 0; i < 4; i++) { spawn(); B[i].p = (B[i].v ? H : W) * (0.2 + i * 0.18); }
  draw(); run();
})();
